🚀 FRESH VPS DEPLOYMENT (First Time)
───────────────────────────────────────────────────────────────

# 1. Clone and prepare
cd /var/www
git clone <YOUR-REPO-URL> technurture-lms
cd technurture-lms
nano .env.production    # Add DATABASE_URL, REDIS_URL, NODE_ENV=production

# 2. Install and build
rm -rf node_modules .next
npm install --production
npm run build

# 3. Database (ONLY first time)
npm run db:push
npm run db:seed

# 4. Start with PM2
npm install -g pm2
pm2 start npm --name "technurture-lms" -- start
pm2 startup
pm2 save

# 5. Verify
pm2 status
pm2 logs technurture-lms


🔄 UPDATE DEPLOYMENT (After First Setup)
───────────────────────────────────────────────────────────────

# Pull new code
git pull origin main

# Update deps (if package.json changed)
npm install --production

# Build ONLY (NO db:seed!)
npm run build

# Restart app safely
pm2 restart technurture-lms

# Verify
pm2 status
pm2 logs technurture-lms


🛑 DO NOT RUN (These Delete Data!)
───────────────────────────────────────────────────────────────

❌ npm run setup
❌ npm run db:seed  (except first-time setup)
❌ npm run db:migrate  (without backup)
❌ docker-compose down


✅ VERIFY DEPLOYMENT
───────────────────────────────────────────────────────────────

# Check app running
pm2 status

# View logs for errors
pm2 logs technurture-lms

# Test API
curl http://localhost:3000

# Check database connection
npm run db:studio


📞 COMMON ISSUES
───────────────────────────────────────────────────────────────

App won't start?
  pm2 logs technurture-lms
  lsof -i :3000
  pm2 restart technurture-lms

Database error?
  systemctl status postgresql
  cat .env.production | grep DATABASE_URL

High memory?
  pm2 restart technurture-lms
  pm2 monit
